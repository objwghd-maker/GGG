local function loadAsset(fileName, url)
    local assetSuccess, assetResult = pcall(function()
        if type(isfile) == "function" and type(writefile) == "function" and type(getcustomasset) == "function" then
            if not isfile(fileName) then
                local data = game:HttpGet(url)
                if data and #data > 0 then
                    if type(crypt) == "table" and crypt.base64 and type(crypt.base64.decode) == "function" then
                        writefile(fileName, crypt.base64.decode(data))
                    else
                        writefile(fileName, data)
                    end
                end
            end
            return getcustomasset(fileName)
        end
        return ""
    end)
    if assetSuccess and assetResult then return assetResult end
    return ""
end

local function createSpikeEffect(parent, name, size, radius, count)
    local effect = Instance.new("Frame")
    effect.Name = name
    effect.AnchorPoint = Vector2.new(0.5, 0.5)
    effect.Size = UDim2.fromOffset(size, size)
    effect.BackgroundTransparency = 1
    effect.BorderSizePixel = 0
    effect.ClipsDescendants = false
    effect.ZIndex = 7
    effect.Parent = parent

    local center = size / 2

    for i = 1, count do
        local angle = (i - 1) * 360 / count

        local spike = Instance.new("TextLabel")
        spike.Name = "RedSpike_" .. i
        spike.AnchorPoint = Vector2.new(0.5, 0.5)
        spike.Size = UDim2.fromOffset(12, 16)
        spike.Position = UDim2.fromOffset(
            center + math.cos(math.rad(angle)) * radius,
            center + math.sin(math.rad(angle)) * radius
        )
        spike.Rotation = angle + 90
        spike.BackgroundTransparency = 1
        spike.Text = "▲"
        spike.TextColor3 = Color3.fromRGB(255, 20, 35)
        spike.TextStrokeColor3 = Color3.fromRGB(100, 0, 0)
        spike.TextStrokeTransparency = 0
        spike.TextSize = 16
        spike.Font = Enum.Font.GothamBlack
        spike.ZIndex = 7
        spike.Parent = effect

        TweenService:Create(
            spike,
            TweenInfo.new(0.7 + (i % 4) * 0.1, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
            {
                TextColor3 = Color3.fromRGB(255, 100, 115),
                TextSize = 19,
                TextTransparency = 0.1
            }
        ):Play()
    end

    TweenService:Create(
        effect,
        TweenInfo.new(7, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1),
        {Rotation = 360}
    ):Play()

    return effect
end

RunService.RenderStepped:Connect(function()
    if not Enabled then return end

    if currentTarget and currentTarget.Parent then
        local humanoid = currentTarget:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then
            currentTarget = nil
        end
    else
        currentTarget = nil
    end

    if not currentTarget then
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Model") then
                local humanoid = obj:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    if targetNames[obj.Name] then
                        local rootPart = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Head")
                        if rootPart then
                            currentTarget = obj
                            break
                        end
                    end
                end
            end
        end
    end

    if currentTarget and Camera then
        local humanoid = currentTarget:FindFirstChildOfClass("Humanoid")
        if humanoid then
            Camera.CameraSubject = humanoid
        end
    end
end)

    Instance.new("UICorner", textBox).CornerRadius = UDim.new(0, 4)

    local textBoxStroke = Instance.new("UIStroke")

    textBoxStroke.Parent = textBox

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -20, 0, 6)
    track.Position = UDim2.new(0, 10, 0, 34)
    track.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    track.BorderSizePixel = 0
    track.ZIndex = 5
    track.Parent = row

    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((defaultVal - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
    fill.BorderSizePixel = 0
    fill.ZIndex = 5
    fill.Parent = track

    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local knob = Instance.new("TextButton")
    knob.Size = UDim2.fromOffset(16, 16)
    knob.AnchorPoint = Vector2.new(0.5, 0.5)
    knob.Position = UDim2.new((defaultVal - min) / (max - min), 0, 0.5, 0)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.Text = ""
    knob.ZIndex = 6
    knob.Parent = track

    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

    local sliding = false

    local function updateValue(val)
        val = math.clamp(val, min, max)
        local ratio = (val - min) / (max - min)
        fill.Size = UDim2.new(ratio, 0, 1, 0)
        knob.Position = UDim2.new(ratio, 0, 0.5, 0)
        textBox.Text = tostring(math.floor(val))
        callback(math.floor(val))
    end

    knob.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local pos = input.Position.X
            local absPos = track.AbsolutePosition.X
            local absSize = track.AbsoluteSize.X
            local ratio = math.clamp((pos - absPos) / absSize, 0, 1)
            local val = min + ratio * (max - min)
            updateValue(val)
        end
    end)

    textBox.FocusLost:Connect(function()
        local num = tonumber(textBox.Text)
        if num then
            updateValue(num)
        else
            textBox.Text = tostring(defaultVal)
        end
    end)
end

local dragging = false
local dragStart
local startPos
local miniDragging = false
local miniDragStart
local miniStartPos

local function updateResponsiveSize()
    local camera = workspace.CurrentCamera
    if not camera then return end

    local viewport = camera.ViewportSize
    local margin = 10
    local width
    local height

    if viewport.X > viewport.Y then
        width = math.min(viewport.X * 0.42, 380)
        height = width * 0.62
    else
        width = math.min(viewport.X * 0.80, 350)
        height = width * 0.62
    end

    width = math.min(width, viewport.X - margin * 2)
    height = math.min(height, viewport.Y - margin * 2)

    MainFrame.Size = UDim2.fromOffset(width, height)

    if not dragging then
        MainFrame.AnchorPoint = Vector2.new(0.5, 0.5)
        MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
    else
        local x = math.clamp(MainFrame.Position.X.Offset, width / 2, viewport.X - width / 2)
        local y = math.clamp(MainFrame.Position.Y.Offset, height / 2, viewport.Y - height / 2)
        MainFrame.Position = UDim2.fromOffset(x, y)
    end

    local miniSize = MinimizedIcon.AbsoluteSize
    MinimizedIcon.Position = UDim2.fromOffset(
        math.clamp(MinimizedIcon.Position.X.Offset, 0, math.max(0, viewport.X - miniSize.X)),
        math.clamp(MinimizedIcon.Position.Y.Offset, 0, math.max(0, viewport.Y - miniSize.Y))
    )
end

local cameraConnection

local function bindCamera()
    if cameraConnection then
        cameraConnection:Disconnect()
        cameraConnection = nil
    end

    local camera = workspace.CurrentCamera
    if camera then
        cameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateResponsiveSize)
    end
end

updateResponsiveSize()
bindCamera()

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    bindCamera()
    updateResponsiveSize()
end)

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        local camera = workspace.CurrentCamera

        if camera then
            local viewport = camera.ViewportSize
            local size = MainFrame.AbsoluteSize
            local x = math.clamp(startPos.X.Offset + delta.X, size.X / 2, viewport.X - size.X / 2)
            local y = math.clamp(startPos.Y.Offset + delta.Y, size.Y / 2, viewport.Y - size.Y / 2)

            MainFrame.AnchorPoint = Vector2.new(0.5, 0.5)
            MainFrame.Position = UDim2.fromOffset(x, y)
        end
    end
end)

TopBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
